1//! bizstd — "Blazing-fast Insert ZSTD compressed file format".
2//!
3//! A self-describing binary container: a fixed 16-byte preamble, a
4//! fixed-size text header zone updated in place, and a data section of
5//! independent zstd frames followed by an uncompressed crash-safe tail.
6//!
7//! ```text
8//! preamble (16 bytes, binary):
9//!   magic "BIZSTD"   6 bytes
10//!   version          u8
11//!   flags            u8      (bit 0: data compressed)
12//!   header_area_len  u32 LE  (usually 4096)
13//!   reserved         4 bytes
14//!
15//! header zone (ASCII text inside header_area_len bytes):
16//!   key:value\n      key `_?[a-z0-9][a-z0-9_]*`, split on the FIRST ':'
17//!   \n               an empty line ends the headers
18//!   zero padding to the end of the zone
19//!
20//! data (at offset 16 + header_area_len):
21//!   zstd frames of closed hours, then the raw uncompressed tail of the
22//!   hour being written
23//! ```
24//!
25//! System keys start with `_` and are a closed list owned by this file:
26//! `_schema`, `_schema_fields`, `_schema_hash`, `_record`, `_source`,
27//! `_created_at`, `_writer`, `_compression`, `_compression_level`,
28//! `_records`, `_bytes_raw`, `_frames`, `_preview`, `_sealed`. Every one of
29//! them is a rebuildable cache — the data section is the truth. The
30//! counters `_records` and `_bytes_raw` cover closed frames only; records
31//! still sitting in the raw tail join them when their frame closes.
32//! Application keys must NOT start with `_`; the container never
33//! interprets them.
34//!
35//! Values escape exactly two characters: `\n` and `\\`. Header values built
36//! by this file contain neither, so their round trip is byte-identical.
37//!
38//! Closing a frame is crash-safe through a redo journal (`<file>.seal`):
39//! the compressed frame and the new header zone become durable in the
40//! journal first, then the raw tail is overwritten in place, the file is
41//! truncated and the zone rewritten, and only then the journal is removed.
42//! [`Container::open_append`] replays a complete journal and discards an
43//! incomplete one, then truncates a torn record off the raw tail — a crash
44//! at any instant loses at most the buffered records of the current write.
45//!
46//! The crate is deliberately small — `std` and `zstd` only, no other
47//! dependency and nothing to configure — so that reading a container needs
48//! this crate and nothing else.
49//!
50//! # Concurrency
51//!
52//! **One writer per file, and enforcing that is the caller's job.** Nothing
53//! here takes a lock. Two handles opened with [`Container::open_append`] on
54//! the same path will both recover the tail, both buffer records, and both
55//! write at offsets the other does not know about; the file that results is
56//! not one either of them wrote. There is no error for this and no detection —
57//! only the guarantee that it will not be noticed until the data is read back.
58//!
59//! Readers are unrestricted among themselves: [`Container::open_read`] never
60//! writes. A reader running alongside a writer is safe in the sense that it
61//! cannot corrupt anything, but what it sees is a file in motion — the tail
62//! grows under it, and a frame being closed changes both the data and the
63//! header zone. Reading a file that is being written to gives a consistent
64//! answer only for the frames already closed when the read started.
65//!
66//! # Example
67//!
68//! ```no_run
69//! use bizstd::{Container, DEFAULT_HEADER_AREA, FieldSpec, HOT_LEVEL, RecordLayout, Schema};
70//!
71//! # fn main() -> bizstd::Result<()> {
72//! let schema = Schema {
73//!     name: "samples@1".to_owned(),
74//!     fields: vec![
75//!         FieldSpec { name: "time_nanos".to_owned(), ty: "u64".to_owned(), offset: 0 },
76//!         FieldSpec { name: "value".to_owned(), ty: "f64".to_owned(), offset: 8 },
77//!     ],
78//!     layout: RecordLayout::Fixed(16),
79//! };
80//!
81//! let mut file = Container::create(
82//!     "samples.bizstd".as_ref(),
83//!     &schema,
84//!     "example",
85//!     "example writer",
86//!     0,
87//!     DEFAULT_HEADER_AREA,
88//!     &[("stream", "alpha")],
89//! )?;
90//!
91//! let mut record = [0u8; 16];
92//! record[..8].copy_from_slice(&1_u64.to_le_bytes());
93//! record[8..].copy_from_slice(&2.5_f64.to_le_bytes());
94//! file.append_record(&record)?;
95//! file.seal(0, HOT_LEVEL)?;
96//! # Ok(())
97//! # }
98//! ```

110/// Everything this crate can fail with.
111///
112/// The variants are the distinctions a caller can act on: a malformed file is
113/// worth quarantining, a full header zone is worth repacking, and a usage
114/// mistake is worth fixing in the calling code. Marked `#[non_exhaustive]` so
115/// that a later version can name a new failure without breaking the callers
116/// that match on this one.
117#[derive(Debug)]
118#[non_exhaustive]
119pub enum Error {
120    /// An I/O failure with the path it happened on.
121    Io {
122        /// The file involved, when one is known.
123        path: Option<PathBuf>,
124        /// What the operating system said.
125        source: std::io::Error,
126    },
127    /// The file is not a well-formed container: bad magic, unknown version,
128    /// unparsable headers, a frame list that does not match the bytes.
129    Malformed(String),
130    /// The caller asked for something the container cannot do: writing to a
131    /// read-only handle, a record that does not match the schema, a header
132    /// key that is not theirs to set.
133    Usage(String),
134    /// The header zone cannot hold the headers. Distinguishable on purpose:
135    /// it is the one failure a long-lived writer hits by growing normally,
136    /// and the way out is [`repack_with_header_area`], not a code change.
137    ZoneFull {
138        /// Bytes the headers need.
139        needed: usize,
140        /// Bytes the zone holds.
141        capacity: usize,
142    },
143    /// zstd refused to compress or decompress.
144    Compression(String),
145}

192/// Refuses a header zone the file is not big enough to hold.
193///
194/// Both numbers come from the file, so believing either one on its own is how
195/// a twenty-byte file asks for a gigabyte.
196fn check_zone_fits(preamble: Preamble, file_len: u64) -> Result<()> {
197    let needed = (PREAMBLE_BYTES as u64).saturating_add(u64::from(preamble.header_area));
198    if file_len < needed {
199        return Err(malformed(format!(
200            "header zone claims {} bytes but the file is {file_len} bytes",
201            preamble.header_area
202        )));
203    }
204    Ok(())
205}

207/// Refuses a frame list that points outside the file.
208fn check_frames_fit(frames: &[Frame], data_start: u64, file_len: u64) -> Result<()> {
209    for frame in frames {
210        let end = data_start
211            .checked_add(frame.offset)
212            .and_then(|start| start.checked_add(frame.len))
213            .ok_or_else(|| malformed(format!("frame {} overflows a u64 offset", frame.id)))?;
214        if end > file_len {
215            return Err(malformed(format!("frame {} ends at {end} but the file is {file_len} bytes", frame.id)));
216        }
217    }
218    Ok(())
219}

263/// FNV-1a 64 — the schema fingerprint. A fingerprint, not a signature.
264#[must_use]
265pub fn fnv1a64(bytes: &[u8]) -> u64 {
266    let mut hash: u64 = 0xCBF2_9CE4_8422_2325;
267    for byte in bytes {
268        hash ^= u64::from(*byte);
269        hash = hash.wrapping_mul(0x0000_0100_0000_01B3);
270    }
271    hash
272}

274/// Whether the text is a legal header key: `_?[a-z0-9][a-z0-9_]*`.
275#[must_use]
276pub fn key_is_valid(key: &str) -> bool {
277    let body = key.strip_prefix('_').unwrap_or(key);
278    let mut chars = body.chars();
279    let Some(first) = chars.next() else {
280        return false;
281    };
282    if !first.is_ascii_lowercase() && !first.is_ascii_digit() {
283        return false;
284    }
285    chars.all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == '_')
286}

288/// Escapes the two characters a value may not carry raw: `\n` and `\`.
289#[must_use]
290pub fn escape_value(value: &str) -> String {
291    let mut out = String::with_capacity(value.len());
292    for c in value.chars() {
293        match c {
294            '\\' => out.push_str("\\\\"),
295            '\n' => out.push_str("\\n"),
296            other => out.push(other),
297        }
298    }
299    out
300}
301
302/// Reverses [`escape_value`].
303#[must_use]
304pub fn unescape_value(value: &str) -> String {
305    let mut out = String::with_capacity(value.len());
306    let mut chars = value.chars();
307    while let Some(c) = chars.next() {
308        if c == '\\' {
309            match chars.next() {
310                Some('n') => out.push('\n'),
311                Some('\\') => out.push('\\'),
312                Some(other) => out.push(other),
313                None => {}
314            }
315        } else {
316            out.push(c);
317        }
318    }
319    out
320}

322/// The 16-byte binary preamble.
323#[derive(Debug, Clone, Copy, PartialEq, Eq)]
324pub struct Preamble {
325    /// Container format version.
326    pub version: u8,
327    /// Bit 0: data compressed.
328    pub flags: u8,
329    /// The header zone size in bytes.
330    pub header_area: u32,
331}
332
333impl Preamble {
334    /// The 16 bytes.
335    #[must_use]
336    pub fn encode(self) -> [u8; PREAMBLE_BYTES] {
337        let mut out = [0u8; PREAMBLE_BYTES];
338        out.iter_mut()
339            .zip(MAGIC.iter())
340            .for_each(|(slot, byte)| *slot = *byte);
341        if let Some(slot) = out.get_mut(6) {
342            *slot = self.version;
343        }
344        if let Some(slot) = out.get_mut(7) {
345            *slot = self.flags;
346        }
347        if let Some(area) = out.get_mut(8..12) {
348            area.copy_from_slice(&self.header_area.to_le_bytes());
349        }
350        out
351    }
352
353    /// Reads the preamble back.
354    ///
355    /// # Errors
356    /// Returns a message when the magic, version or zone size is wrong.
357    pub fn decode(bytes: &[u8]) -> Result<Self> {
358        if bytes.get(0..6) != Some(MAGIC.as_slice()) {
359            return Err(malformed("not a bizstd file: bad magic"));
360        }
361        let version = bytes.get(6).copied().ok_or_else(|| malformed("preamble too short"))?;
362        if version != VERSION {
363            return Err(malformed(format!("unsupported bizstd version {version}")));
364        }
365        let flags = bytes.get(7).copied().ok_or_else(|| malformed("preamble too short"))?;
366        let header_area = bytes
367            .get(8..12)
368            .and_then(|four| four.try_into().ok())
369            .map(u32::from_le_bytes)
370            .ok_or_else(|| malformed("preamble too short"))?;
371        if header_area == 0 {
372            return Err(malformed("header zone size is zero"));
373        }
374        if header_area > MAX_HEADER_AREA {
375            return Err(malformed(format!(
376                "header zone claims {header_area} bytes, the ceiling is {MAX_HEADER_AREA}"
377            )));
378        }
379        Ok(Self { version, flags, header_area })
380    }
381}

383/// The ordered key:value pairs of the header zone.
384#[derive(Debug, Clone, Default, PartialEq, Eq)]
385pub struct Headers {
386    pairs: Vec<(String, String)>,
387}
388
389impl Headers {
390    /// The value of a key, unescaped.
391    #[must_use]
392    pub fn get(&self, key: &str) -> Option<&str> {
393        self.pairs
394            .iter()
395            .find(|(name, _value)| name == key)
396            .map(|(_name, value)| value.as_str())
397    }
398
399    /// Sets a key, replacing an existing value, keeping first-set order.
400    ///
401    /// # Errors
402    /// Returns a message when the key breaks the grammar.
403    pub fn set(&mut self, key: &str, value: &str) -> Result<()> {
404        if !key_is_valid(key) {
405            return Err(usage(format!("invalid header key {key:?}")));
406        }
407        match self.pairs.iter_mut().find(|(name, _value)| name == key) {
408            Some((_name, slot)) => value.clone_into(slot),
409            None => self.pairs.push((key.to_owned(), value.to_owned())),
410        }
411        Ok(())
412    }
413
414    /// Sets an application key; the leading `_` is reserved for the container.
415    ///
416    /// # Errors
417    /// Returns a message on a `_`-prefixed or ungrammatical key.
418    pub fn set_user(&mut self, key: &str, value: &str) -> Result<()> {
419        if key.starts_with('_') {
420            return Err(usage(format!("user header key {key:?} may not start with '_'")));
421        }
422        self.set(key, value)
423    }
424
425    /// Every pair in order.
426    #[must_use]
427    pub fn pairs(&self) -> &[(String, String)] {
428        &self.pairs
429    }
430
431    /// Parses a header zone: lines to the first empty line, zeros ignored.
432    ///
433    /// # Errors
434    /// Returns a message on non-UTF-8 text, a missing terminator, or a line
435    /// without a colon.
436    pub fn parse(zone: &[u8]) -> Result<Self> {
437        let text = std::str::from_utf8(zone).map_err(|error| malformed(format!("header zone is not UTF-8: {error}")))?;
438        let mut pairs = Vec::new();
439        let mut terminated = false;
440        for line in text.split('\n') {
441            if line.is_empty() || line.starts_with('\0') {
442                terminated = true;
443                break;
444            }
445            let (key, value) = line
446                .split_once(':')
447                .ok_or_else(|| malformed(format!("header line without ':': {line:?}")))?;
448            if !key_is_valid(key) {
449                return Err(usage(format!("invalid header key {key:?}")));
450            }
451            pairs.push((key.to_owned(), unescape_value(value)));
452        }
453        if !terminated {
454            return Err(malformed("header zone has no empty-line terminator"));
455        }
456        Ok(Self { pairs })
457    }
458
459    /// Renders the zone: escaped lines, an empty line, zero padding.
460    ///
461    /// # Errors
462    /// Returns a message when the text does not fit the zone.
463    pub fn render(&self, area: usize) -> Result<Vec<u8>> {
464        let mut text = String::new();
465        for (key, value) in &self.pairs {
466            text.push_str(key);
467            text.push(':');
468            text.push_str(&escape_value(value));
469            text.push('\n');
470        }
471        text.push('\n');
472        if text.len() > area {
473            return Err(Error::ZoneFull { needed: text.len(), capacity: area });
474        }
475        let mut zone = text.into_bytes();
476        zone.resize(area, 0);
477        Ok(zone)
478    }
479}

481/// How record boundaries are found in the raw byte stream.
482#[derive(Debug, Clone, Copy, PartialEq, Eq)]
483pub enum RecordLayout {
484    /// Every record is exactly this many bytes.
485    Fixed(u32),
486    /// Every record is a u16 LE body length followed by the body.
487    Prefixed,
488}
489
490impl RecordLayout {
491    /// The `_record` header value.
492    #[must_use]
493    pub fn render(self) -> String {
494        match self {
495            Self::Fixed(bytes) => format!("fixed:{bytes}"),
496            Self::Prefixed => "prefixed".to_owned(),
497        }
498    }
499
500    /// Parses the `_record` header value.
501    ///
502    /// # Errors
503    /// Returns a message on an unknown shape or a zero size.
504    pub fn parse(text: &str) -> Result<Self> {
505        if text == "prefixed" {
506            return Ok(Self::Prefixed);
507        }
508        if let Some(size) = text.strip_prefix("fixed:") {
509            let bytes: u32 = size
510                .parse()
511                .map_err(|error| malformed(format!("bad record size {size:?}: {error}")))?;
512            if bytes == 0 {
513                return Err(malformed("record size is zero"));
514            }
515            return Ok(Self::Fixed(bytes));
516        }
517        Err(malformed(format!("unknown record layout {text:?}")))
518    }
519}

532/// What the records are: a name, the fields, and the boundary rule.
533#[derive(Debug, Clone, PartialEq, Eq)]
534pub struct Schema {
535    /// The `_schema` value, e.g. `samples@1`.
536    pub name: String,
537    /// The fields in offset order.
538    pub fields: Vec<FieldSpec>,
539    /// How record boundaries are found.
540    pub layout: RecordLayout,
541}
542
543impl Schema {
544    /// The `_schema_fields` value; `_schema_hash` is FNV-1a 64 of exactly
545    /// this string, so no separate canonical form exists.
546    #[must_use]
547    pub fn fields_value(&self) -> String {
548        self.fields
549            .iter()
550            .map(|field| format!("{}:{}:{}", field.name, field.ty, field.offset))
551            .collect::<Vec<_>>()
552            .join(";")
553    }
554
555    /// The `_schema_hash` value.
556    #[must_use]
557    pub fn hash_hex(&self) -> String {
558        format!("{:016x}", fnv1a64(self.fields_value().as_bytes()))
559    }
560
561    /// Reads the schema back from parsed headers.
562    ///
563    /// # Errors
564    /// Returns a message when a schema header is missing or malformed.
565    pub fn from_headers(headers: &Headers) -> Result<Self> {
566        let name = headers
567            .get("_schema")
568            .ok_or_else(|| malformed("missing _schema"))?
569            .to_owned();
570        let layout = RecordLayout::parse(headers.get("_record").ok_or_else(|| malformed("missing _record"))?)?;
571        let mut fields = Vec::new();
572        let fields_value = headers
573            .get("_schema_fields")
574            .ok_or_else(|| malformed("missing _schema_fields"))?;
575        for token in fields_value.split(';').filter(|token| !token.is_empty()) {
576            let mut parts = token.splitn(3, ':');
577            let field_name = parts.next().unwrap_or_default();
578            let ty = parts.next().unwrap_or_default();
579            let offset: u32 = parts
580                .next()
581                .unwrap_or_default()
582                .parse()
583                .map_err(|error| malformed(format!("bad field token {token:?}: {error}")))?;
584            if field_name.is_empty() || ty.is_empty() {
585                return Err(malformed(format!("bad field token {token:?}")));
586            }
587            fields.push(FieldSpec { name: field_name.to_owned(), ty: ty.to_owned(), offset });
588        }
589        Ok(Self { name, fields, layout })
590    }

593/// One closed zstd frame in the data section.
594#[derive(Debug, Clone, Copy, PartialEq, Eq)]
595pub struct Frame {
596    /// The caller's frame id — for time-partitioned data, the period index.
597    pub id: u64,
598    /// Byte offset relative to the start of the data section.
599    pub offset: u64,
600    /// Compressed length in bytes.
601    pub len: u64,
602    /// XXH64 of the compressed bytes, `0` when the file predates checksums.
603    /// Zero is a possible-but-astronomically-unlikely real hash; treating it

606    pub hash: u64,

609/// Renders `_frames`: `id,offset,len,xxh64;...` (hash in hex; the fourth
610/// field is omitted while unrecorded, so an index written before checksums
611/// existed round-trips byte-for-byte).
612#[must_use]
613pub fn render_frames(frames: &[Frame]) -> String {
614    frames
615        .iter()
616        .map(|frame| {
617            if frame.hash == 0 {
618                format!("{},{},{}", frame.id, frame.offset, frame.len)
619            } else {
620                format!("{},{},{},{:016x}", frame.id, frame.offset, frame.len, frame.hash)
621            }
622        })
623        .collect::<Vec<_>>()
624        .join(";")
625}

627/// Parses `_frames`. The fourth field, the frame checksum, is optional:
628/// files written before it existed carry triples.
629///
630/// # Errors
631/// Returns a message on a malformed triple.
632pub fn parse_frames(text: &str) -> Result<Vec<Frame>> {
633    let mut frames = Vec::new();
634    for token in text.split(';').filter(|token| !token.is_empty()) {
635        let mut parts = token.splitn(4, ',');
636        let id: u64 = parts
637            .next()
638            .unwrap_or_default()
639            .parse()
640            .map_err(|error| malformed(format!("bad frame {token:?}: {error}")))?;
641        let offset: u64 = parts
642            .next()
643            .unwrap_or_default()
644            .parse()
645            .map_err(|error| malformed(format!("bad frame {token:?}: {error}")))?;
646        let len: u64 = parts
647            .next()
648            .unwrap_or_default()
649            .parse()
650            .map_err(|error| malformed(format!("bad frame {token:?}: {error}")))?;
651        let hash = match parts.next() {
652            None | Some("") => 0,
653            Some(hex) => u64::from_str_radix(hex, 16).map_err(|error| malformed(format!("bad frame {token:?}: {error}")))?,
654        };
655        frames.push(Frame { id, offset, len, hash });
656    }
657    Ok(frames)
658}

660/// XXH64 of `data` with seed 0 — the per-frame checksum.
661///
662/// Written out rather than pulled from a dependency: it is forty lines, and
663/// the alternative is a second dependency in a crate whose selling point is
664/// having one. Constants and steps follow the xxHash specification, and the
665/// test vectors pin the implementation to it.
666#[must_use]
667pub fn xxh64(data: &[u8]) -> u64 {
668    const P1: u64 = 0x9E37_79B1_85EB_CA87;
669    const P2: u64 = 0xC2B2_AE3D_27D4_EB4F;
670    const P3: u64 = 0x1656_67B1_9E37_79F9;
671    const P4: u64 = 0x85EB_CA77_C2B2_AE63;
672    const P5: u64 = 0x27D4_EB2F_1656_67C5;
673
674    let read_u64 = |chunk: &[u8]| -> u64 { chunk.try_into().map_or(0, u64::from_le_bytes) };
675    let round = |accumulator: u64, lane: u64| -> u64 {
676        accumulator
677            .wrapping_add(lane.wrapping_mul(P2))
678            .rotate_left(31)
679            .wrapping_mul(P1)
680    };
681
682    let mut remainder = data;
683    let mut hash = if data.len() >= 32 {
684        let mut v1 = P1.wrapping_add(P2);
685        let mut v2 = P2;
686        let mut v3 = 0u64;
687        let mut v4 = 0u64.wrapping_sub(P1);
688        while remainder.len() >= 32 {
689            let (block, rest) = remainder.split_at(32);
690            v1 = round(v1, read_u64(block.get(0..8).unwrap_or_default()));
691            v2 = round(v2, read_u64(block.get(8..16).unwrap_or_default()));
692            v3 = round(v3, read_u64(block.get(16..24).unwrap_or_default()));
693            v4 = round(v4, read_u64(block.get(24..32).unwrap_or_default()));
694            remainder = rest;
695        }
696        let mut merged = v1
697            .rotate_left(1)
698            .wrapping_add(v2.rotate_left(7))
699            .wrapping_add(v3.rotate_left(12))
700            .wrapping_add(v4.rotate_left(18));
701        for lane in [v1, v2, v3, v4] {
702            merged = (merged ^ round(0, lane)).wrapping_mul(P1).wrapping_add(P4);
703        }
704        merged
705    } else {
706        P5
707    };
708    hash = hash.wrapping_add(data.len() as u64);
709
710    while remainder.len() >= 8 {
711        let (chunk, rest) = remainder.split_at(8);
712        hash = (hash ^ round(0, read_u64(chunk)))
713            .rotate_left(27)
714            .wrapping_mul(P1)
715            .wrapping_add(P4);
716        remainder = rest;
717    }
718    if remainder.len() >= 4 {
719        let (chunk, rest) = remainder.split_at(4);
720        let lane = u64::from(chunk.try_into().map_or(0, u32::from_le_bytes));
721        hash = (hash ^ lane.wrapping_mul(P1))
722            .rotate_left(23)
723            .wrapping_mul(P2)
724            .wrapping_add(P3);
725        remainder = rest;
726    }
727    for byte in remainder {
728        hash = (hash ^ u64::from(*byte).wrapping_mul(P5))
729            .rotate_left(11)
730            .wrapping_mul(P1);
731    }
732
733    hash ^= hash >> 33;
734    hash = hash.wrapping_mul(P2);
735    hash ^= hash >> 29;
736    hash = hash.wrapping_mul(P3);
737    hash ^= hash >> 32;
738    hash
739}

741/// Compresses one frame with the zstd frame checksum enabled: four bytes of
742/// XXH64 inside the frame, verified by every decoder that reads it.
743fn compress_frame(raw: &[u8], level: i32) -> Result<Vec<u8>> {
744    let mut encoder = zstd::stream::Encoder::new(Vec::new(), level).map_err(Error::from)?;
745    encoder.include_checksum(true).map_err(Error::from)?;
746    encoder.write_all(raw).map_err(Error::from)?;
747    encoder.finish().map_err(Error::from)
748}

750/// Splits raw bytes into records; the second value is the torn leftover.
751#[must_use]
752pub fn split_records(bytes: &[u8], layout: RecordLayout) -> (Vec<&[u8]>, usize) {
753    match layout {
754        RecordLayout::Fixed(size) => {
755            let size = size as usize;
756            if size == 0 {
757                return (Vec::new(), bytes.len());
758            }
759            let records: Vec<&[u8]> = bytes.chunks_exact(size).collect();
760            let leftover = bytes.len().checked_rem(size).unwrap_or(0);
761            (records, leftover)
762        }
763        RecordLayout::Prefixed => {
764            let mut records = Vec::new();
765            let mut position = 0usize;
766            while let Some(prefix) = bytes
767                .get(position..position.saturating_add(2))
768                .and_then(|two| two.try_into().ok())
769                .map(u16::from_le_bytes)
770            {
771                let body_start = position.saturating_add(2);
772                let body_end = body_start.saturating_add(prefix as usize);
773                // The BODY, without its length prefix: a record's fields sit
774                // at the schema's offsets, and a slice that still carried the
775                // two prefix bytes shifted every field read by two.
776                let Some(record) = bytes.get(body_start..body_end) else {
777                    break;
778                };
779                records.push(record);
780                position = body_end;
781            }
782            (records, bytes.len().saturating_sub(position))
783        }
784    }
785}

787/// RFC 3339 UTC from epoch milliseconds — civil-from-days, Hinnant.
788#[must_use]
789pub fn rfc3339_utc(millis: u64) -> String {
790    let seconds = millis.checked_div(1000).unwrap_or(0);
791    let days = i64::try_from(seconds.checked_div(86_400).unwrap_or(0)).unwrap_or(0);
792    let second_of_day = seconds.checked_rem(86_400).unwrap_or(0);
793    #[expect(
794        clippy::integer_division,
795        reason = "the civil-calendar algorithm is defined in integer divisions"
796    )]
797    let (year, month, day) = {
798        let shifted = days.saturating_add(719_468);
799        let era = shifted.div_euclid(146_097);
800        let day_of_era = shifted.rem_euclid(146_097);
801        let year_of_era = day_of_era
802            .saturating_sub(day_of_era / 1460)
803            .saturating_add(day_of_era / 36_524)
804            .saturating_sub(day_of_era / 146_096)
805            / 365;
806        let day_of_year = day_of_era.saturating_sub(
807            365i64
808                .saturating_mul(year_of_era)
809                .saturating_add(year_of_era / 4)
810                .saturating_sub(year_of_era / 100),
811        );
812        let month_index = (5i64.saturating_mul(day_of_year).saturating_add(2)) / 153;
813        let day = day_of_year
814            .saturating_sub((153i64.saturating_mul(month_index).saturating_add(2)) / 5)
815            .saturating_add(1);
816        let month = if month_index < 10 {
817            month_index.saturating_add(3)
818        } else {
819            month_index.saturating_sub(9)
820        };
821        let year = year_of_era
822            .saturating_add(era.saturating_mul(400))
823            .saturating_add(i64::from(month <= 2));
824        (year, month, day)
825    };
826    let hour = second_of_day.checked_div(3600).unwrap_or(0);
827    let minute = second_of_day
828        .checked_rem(3600)
829        .unwrap_or(0)
830        .checked_div(60)
831        .unwrap_or(0);
832    let second = second_of_day.checked_rem(60).unwrap_or(0);
833    format!("{year:04}-{month:02}-{day:02}T{hour:02}:{minute:02}:{second:02}Z")
834}

1035    /// Appends one record body. `Prefixed` layouts get the u16 LE length
1036    /// written by the container; `Fixed` bodies must match the size exactly.
1037    ///
1038    /// # Errors
1039    /// Returns a message on a size mismatch or a write failure.

1041        if !self.writable {
1042            return Err(usage("file is open read-only"));
1043        }
1044        match self.schema.layout {
1045            RecordLayout::Fixed(size) => {
1046                if body.len() != size as usize {
1047                    return Err(usage(format!("record is {} bytes, the schema says {size}", body.len())));
1048                }
1049                self.pending.extend_from_slice(body);
1050                self.bytes_raw = self.bytes_raw.saturating_add(body.len() as u64);
1051            }
1052            RecordLayout::Prefixed => {
1053                let len = u16::try_from(body.len()).map_err(|_error| usage(format!("record of {} bytes overflows the u16 prefix", body.len())))?;
1054                self.pending.extend_from_slice(&len.to_le_bytes());
1055                self.pending.extend_from_slice(body);
1056                self.bytes_raw = self.bytes_raw.saturating_add(body.len() as u64).saturating_add(2);
1057            }
1058        }
1059        self.records = self.records.saturating_add(1);
1060        if self.pending.len() >= WRITE_BUFFER_BYTES {
1061            self.flush_data()?;
1062        }
1063        Ok(())
1064    }

1081    /// Closes the raw tail into one zstd frame, crash-safe.
1082    ///
1083    /// The compressed frame and the new header zone go to the `<file>.seal`
1084    /// journal first; then the tail is overwritten in place, the file
1085    /// truncated and the zone rewritten; then the journal is removed. A
1086    /// crash anywhere is repaired by [`Container::open_append`]. An empty
1087    /// tail is a no-op.
1088    ///
1089    /// # Errors
1090    /// Returns a message on any write failure.
1091    pub fn close_frame(&mut self, frame_id: u64, level: i32) -> Result<()> {
1092        if !self.writable {
1093            return Err(usage("file is open read-only"));
1094        }
1095        self.flush_data()?;
1096        let tail_start = self.tail_start();
1097        let raw_len = self.file_len.saturating_sub(tail_start);
1098        if raw_len == 0 {
1099            return Ok(());
1100        }
1101        let mut raw = vec![0u8; to_usize(raw_len)?];
1102        self.file.seek(SeekFrom::Start(tail_start)).map_err(Error::from)?;
1103        self.file.read_exact(&mut raw).map_err(Error::from)?;
1104
1105        let preview = if self.headers.get("_preview").unwrap_or_default().is_empty() {
1106            Some(self.render_preview(&raw))
1107        } else {
1108            None
1109        };
1110
1111        let compressed = compress_frame(&raw, level)?;
1112
1113        // Everything below is assembled on copies. A frame pushed onto
1114        // `self.frames` before the zone is known to render would move
1115        // `tail_start()` past bytes the file does not have, and every later
1116        // write would land at the wrong offset — one recoverable error turning
1117        // the handle into a source of corruption for the rest of its life.
1118        let frame = Frame {
1119            id: frame_id,
1120            offset: tail_start.saturating_sub(self.data_start),
1121            len: compressed.len() as u64,
1122            hash: xxh64(&compressed),
1123        };
1124        let mut frames = self.frames.clone();
1125        frames.push(frame);
1126        let records_closed = self.records;
1127        let bytes_closed = self.bytes_raw;
1128        let mut headers = self.headers.clone();
1129        if let Some(preview) = preview {
1130            headers.set("_preview", &preview)?;
1131        }
1132        headers.set("_frames", &render_frames(&frames))?;
1133        headers.set("_records", &records_closed.to_string())?;
1134        headers.set("_bytes_raw", &bytes_closed.to_string())?;
1135        headers.set("_compression_level", &level.to_string())?;
1136        let zone = headers.render(self.preamble.header_area as usize)?;
1137
1138        write_journal(&self.path, tail_start, &zone, &compressed)?;
1139        splice(&mut self.file, tail_start, &zone, &compressed)?;
1140
1141        // Durable. Only now does the in-memory view move.
1142        self.frames = frames;
1143        self.headers = headers;
1144        self.records_closed = records_closed;
1145        self.bytes_closed = bytes_closed;
1146        self.file_len = tail_start.saturating_add(compressed.len() as u64);
1147        remove_journal(&self.path);
1148        Ok(())
1149    }

1176            return Err(usage("file is open read-only"));
1177        }
1178        self.headers.set("_records", &self.records_closed.to_string())?;
1179        self.headers.set("_bytes_raw", &self.bytes_closed.to_string())?;
1180        let zone = self.headers.render(self.preamble.header_area as usize)?;
1181        self.file
1182            .seek(SeekFrom::Start(PREAMBLE_BYTES as u64))
1183            .map_err(Error::from)?;
1184        self.file.write_all(&zone).map_err(Error::from)?;
1185        self.file.sync_data().map_err(Error::from)?;
1186        Ok(())
1187    }

1189    /// Decompresses one closed frame.
1190    ///
1191    /// # Errors
1192    /// Returns a message on an unknown id or a decode failure.
1193    pub fn read_frame(&mut self, frame_id: u64) -> Result<Vec<u8>> {
1194        let frame = self
1195            .frames
1196            .iter()
1197            .find(|frame| frame.id == frame_id)
1198            .copied()
1199            .ok_or_else(|| usage(format!("no frame {frame_id}")))?;
1200        self.read_frame_body(frame)
1201    }

1203    /// Decompresses the frame at a POSITION in the frame list.
1204    ///
1205    /// Frame ids repeat legitimately — a caller that partitions by hour closes
1206    /// a spill under hour 0 after hour 23 — so a sweep meaning "every frame"
1207    /// must go by position. [`Container::read_frame`] stays for callers that
1208    /// mean "the one with this id".
1209    ///
1210    /// # Errors
1211    /// Returns an error on a bad index or a decode failure.
1212    pub fn read_frame_at(&mut self, index: usize) -> Result<Vec<u8>> {
1213        let frame = self
1214            .frames
1215            .get(index)
1216            .copied()
1217            .ok_or_else(|| usage(format!("no frame at position {index}")))?;
1218        self.read_frame_body(frame)
1219    }

1221    fn read_frame_body(&mut self, frame: Frame) -> Result<Vec<u8>> {
1222        check_frames_fit(std::slice::from_ref(&frame), self.data_start, self.file_len)?;
1223        let mut compressed = vec![0u8; to_usize(frame.len)?];
1224        self.file
1225            .seek(SeekFrom::Start(self.data_start.saturating_add(frame.offset)))
1226            .map_err(Error::from)?;
1227        self.file.read_exact(&mut compressed).map_err(Error::from)?;
1228        decode_bounded(&compressed)
1229    }

1231    /// Reads the raw uncompressed tail, whole records only.
1232    ///
1233    /// # Errors
1234    /// Returns the read failure.
1235    pub fn read_tail(&mut self) -> Result<Vec<u8>> {
1236        let tail_start = self.tail_start();
1237        let len = self.file_len.saturating_sub(tail_start);
1238        let mut raw = vec![0u8; to_usize(len)?];
1239        if !raw.is_empty() {
1240            self.file.seek(SeekFrom::Start(tail_start)).map_err(Error::from)?;
1241            self.file.read_exact(&mut raw).map_err(Error::from)?;
1242        }
1243        let (_records, leftover) = split_records(&raw, self.schema.layout);
1244        raw.truncate(raw.len().saturating_sub(leftover));
1245        Ok(raw)
1246    }

1248    /// Renders the first records of the first frame for `_preview`.
1249    fn render_preview(&self, raw: &[u8]) -> String {
1250        let (records, _leftover) = split_records(raw, self.schema.layout);
1251        let mut rendered = Vec::new();
1252        for record in records.iter().take(PREVIEW_RECORDS) {
1253            let mut values = Vec::new();
1254            for field in &self.schema.fields {
1255                values.push(render_field(record, field));
1256            }
1257            rendered.push(values.join(","));
1258        }
1259        let preview = rendered.join(";");
1260        if preview.len() > PREVIEW_BUDGET {
1261            rendered.truncate(1);
1262            return rendered.join(";");
1263        }
1264        preview
1265    }
1266
1267    fn truncate_torn_tail(&mut self) -> Result<()> {
1268        let tail_start = self.tail_start();
1269        let len = self.file_len.saturating_sub(tail_start);
1270        if len == 0 {
1271            return Ok(());
1272        }
1273        let mut raw = vec![0u8; to_usize(len)?];
1274        self.file.seek(SeekFrom::Start(tail_start)).map_err(Error::from)?;
1275        self.file.read_exact(&mut raw).map_err(Error::from)?;
1276        let (records, leftover) = split_records(&raw, self.schema.layout);
1277        if leftover > 0 {
1278            let keep = self.file_len.saturating_sub(leftover as u64);
1279            self.file.set_len(keep).map_err(Error::from)?;
1280            self.file_len = keep;
1281        }
1282        // Header counters stop at the last frame close; the tail on disk is
1283        // what survived after it.
1284        self.records = self
1285            .headers
1286            .get("_records")
1287            .unwrap_or("0")
1288            .parse::<u64>()
1289            .unwrap_or(0)
1290            .saturating_add(records.len() as u64);
1291        // The same correction for bytes. Restoring one counter and not the
1292        // other leaves `_bytes_raw` quietly wrong from the next frame close
1293        // onwards, and nothing checks it until a rebuild.
1294        let surviving = raw.len().saturating_sub(leftover) as u64;
1295        self.bytes_raw = self
1296            .headers
1297            .get("_bytes_raw")
1298            .unwrap_or("0")
1299            .parse::<u64>()
1300            .unwrap_or(0)
1301            .saturating_add(surviving);
1302        Ok(())
1303    }
1304}

1306impl Drop for Container {
1307    /// Writes whatever is still buffered.
1308    ///
1309    /// `append_record` counts a record the moment it is handed over but keeps
1310    /// it in memory until the buffer fills, so a handle dropped on an early
1311    /// return — the most ordinary shape an error path has — used to lose up to
1312    /// the buffer size without a word. Dropping cannot report a failure, so a
1313    /// caller who needs to know still calls `flush_data` or `close_frame`; this
1314    /// only makes the silent case rare instead of routine.
1315    fn drop(&mut self) {
1316        if self.writable && !self.pending.is_empty() {
1317            let _ignored = self.flush_data();
1318        }
1319    }
1320}

1322/// Reads one field of a record for the preview.
1323fn render_field(record: &[u8], field: &FieldSpec) -> String {
1324    let offset = field.offset as usize;
1325    let read = |width: usize| record.get(offset..offset.saturating_add(width));
1326    match field.ty.as_str() {
1327        "u8" => record.get(offset).map(ToString::to_string),
1328        "u16" => read(2)
1329            .and_then(|b| b.try_into().ok())
1330            .map(|b| u16::from_le_bytes(b).to_string()),
1331        "u32" => read(4)
1332            .and_then(|b| b.try_into().ok())
1333            .map(|b| u32::from_le_bytes(b).to_string()),
1334        "u64" => read(8)
1335            .and_then(|b| b.try_into().ok())
1336            .map(|b| u64::from_le_bytes(b).to_string()),
1337        "i64" => read(8)
1338            .and_then(|b| b.try_into().ok())
1339            .map(|b| i64::from_le_bytes(b).to_string()),
1340        "f64" => read(8)
1341            .and_then(|b| b.try_into().ok())
1342            .map(|b| f64::from_le_bytes(b).to_string()),
1343        "uuid" => read(16).map(|bytes| {
1344            bytes.iter().fold(String::with_capacity(32), |mut hex, byte| {
1345                use std::fmt::Write as _;
1346                let _ignored = write!(hex, "{byte:02x}");
1347                hex
1348            })
1349        }),
1350        _other => None,
1351    }
1352    .unwrap_or_else(|| "?".to_owned())
1353}

1355/// The journal path next to a container file.
1356fn journal_path(path: &Path) -> PathBuf {
1357    let mut name = path.file_name().map(ToOwned::to_owned).unwrap_or_default();
1358    name.push(".seal");
1359    path.with_file_name(name)
1360}

1362/// Writes the seal journal durably: magic, payload length, checksum, then
1363/// the tail offset, the zone and the frame bytes.
1364///
1365/// The length and the checksum are what make a torn journal recognisable. A
1366/// crash during this write leaves a prefix on disk, and a prefix is a
1367/// perfectly plausible journal as far as the fields go — without a way to
1368/// tell, replay would splice a truncated frame over a raw tail that is still
1369/// intact, destroying the very records the journal exists to protect.
1370fn write_journal(path: &Path, tail_start: u64, zone: &[u8], compressed: &[u8]) -> Result<()> {
1371    let journal = journal_path(path);
1372    let mut payload = Vec::with_capacity(16usize.saturating_add(zone.len()).saturating_add(compressed.len()));
1373    payload.extend_from_slice(&tail_start.to_le_bytes());
1374    payload.extend_from_slice(&(zone.len() as u64).to_le_bytes());
1375    payload.extend_from_slice(zone);
1376    payload.extend_from_slice(compressed);
1377
1378    let mut body = Vec::with_capacity(JOURNAL_HEAD_BYTES.saturating_add(payload.len()));
1379    body.extend_from_slice(&JOURNAL_MAGIC);
1380    body.extend_from_slice(&(payload.len() as u64).to_le_bytes());
1381    body.extend_from_slice(&fnv1a64(&payload).to_le_bytes());
1382    body.extend_from_slice(&payload);
1383
1384    let mut file = File::create(&journal).map_err(io_at(&journal))?;
1385    file.write_all(&body).map_err(Error::from)?;
1386    file.sync_all().map_err(Error::from)?;
1387    // The bytes are durable; the directory entry that names them is not, and
1388    // on a crash a journal nobody can find is a journal that never existed.
1389    sync_parent_dir(&journal);
1390    Ok(())
1391}
1392
1393/// Flushes the directory entry of a freshly created file.
1394///
1395/// Best effort on purpose: on platforms where a directory cannot be opened
1396/// this does nothing, and the guarantee degrades to "the file's contents are
1397/// durable" rather than failing a write that otherwise succeeded.
1398fn sync_parent_dir(path: &Path) {
1399    // Written as nested `if let` rather than a let-chain on purpose: the chain
1400    // form needs a newer compiler than anything else here does, and the
1401    // language features the code uses are what set the supported version.
1402    if let Some(parent) = path.parent() {
1403        if let Ok(dir) = File::open(parent) {
1404            let _ignored = dir.sync_all();
1405        }
1406    }
1407}

1409/// Overwrites the raw tail with the compressed frame and rewrites the zone.
1410fn splice(file: &mut File, tail_start: u64, zone: &[u8], compressed: &[u8]) -> Result<()> {
1411    file.seek(SeekFrom::Start(tail_start)).map_err(Error::from)?;
1412    file.write_all(compressed).map_err(Error::from)?;
1413    file.set_len(tail_start.saturating_add(compressed.len() as u64))
1414        .map_err(Error::from)?;
1415    file.seek(SeekFrom::Start(PREAMBLE_BYTES as u64))
1416        .map_err(Error::from)?;
1417    file.write_all(zone).map_err(Error::from)?;
1418    file.sync_data().map_err(Error::from)?;
1419    Ok(())
1420}

1426/// Replays a complete seal journal, discards an incomplete one.
1427///
1428/// The journal holds everything the interrupted [`Container::close_frame`]
1429/// was about to make true, so redoing it is idempotent.
1430fn replay_journal(path: &Path) -> Result<()> {
1431    let journal = journal_path(path);
1432    let Ok(body) = std::fs::read(&journal) else {
1433        return Ok(());
1434    };
1435    let parsed = (|| {
1436        if body.get(0..6) != Some(JOURNAL_MAGIC.as_slice()) {
1437            return None;
1438        }
1439        let payload_len = body
1440            .get(6..14)
1441            .and_then(|b| b.try_into().ok())
1442            .map(u64::from_le_bytes)
1443            .and_then(|len| usize::try_from(len).ok())?;
1444        let checksum = body
1445            .get(14..JOURNAL_HEAD_BYTES)
1446            .and_then(|b| b.try_into().ok())
1447            .map(u64::from_le_bytes)?;
1448        // Length first, then checksum: a torn write leaves a short body, and a
1449        // body of the right length can still have been written by a different
1450        // version or scrambled by the filesystem.
1451        let payload = body.get(JOURNAL_HEAD_BYTES..JOURNAL_HEAD_BYTES.checked_add(payload_len)?)?;
1452        if body.len() != JOURNAL_HEAD_BYTES.checked_add(payload_len)? || fnv1a64(payload) != checksum {
1453            return None;
1454        }
1455        let tail_start = payload
1456            .get(0..8)
1457            .and_then(|b| b.try_into().ok())
1458            .map(u64::from_le_bytes)?;
1459        let zone_len = payload
1460            .get(8..16)
1461            .and_then(|b| b.try_into().ok())
1462            .map(u64::from_le_bytes)?;
1463        let zone_end = 16usize.checked_add(usize::try_from(zone_len).ok()?)?;
1464        let zone = payload.get(16..zone_end)?;
1465        let compressed = payload.get(zone_end..)?;
1466        if compressed.is_empty() {
1467            return None;
1468        }
1469        Some((tail_start, zone.to_vec(), compressed.to_vec()))
1470    })();
1471    let Some((tail_start, zone, compressed)) = parsed else {
1472        // Incomplete journal: the crash happened before it became durable,
1473        // so the main file still carries the intact raw tail.
1474        let _ignored = std::fs::remove_file(&journal);
1475        return Ok(());
1476    };
1477    let mut file = OpenOptions::new()
1478        .read(true)
1479        .write(true)
1480        .open(path)
1481        .map_err(io_at(path))?;
1482    splice(&mut file, tail_start, &zone, &compressed)?;
1483    let _ignored = std::fs::remove_file(&journal);
1484    Ok(())
1485}

1493    pub frames: usize,
1494    /// Records seen in frames and the tail.

1498/// Checks a file: headers, schema hash, frame decode, record boundaries,
1499/// counters. `check` receives every record and returns a problem or `None`;
1500/// pass `None` to skip content checks. At most 20 problems are collected.
1501///
1502/// # Errors
1503/// Returns a message when the file cannot be opened at all.
1504pub fn validate(path: &Path, mut check: Option<&mut dyn FnMut(&[u8]) -> Option<String>>) -> Result<ValidateReport> {
1505    // A pending journal is part of the file's state. Reporting on the bytes
1506    // without it describes a file that will look different the moment a writer
1507    // opens it.
1508    replay_journal(path)?;
1509    let mut container = Container::open_read(path)?;
1510    let mut report = ValidateReport::default();
1511    let problem = |report: &mut ValidateReport, text: String| {
1512        if report.problems.len() < 20 {
1513            report.problems.push(text);
1514        }
1515    };
1516
1517    let declared_hash = container.headers.get("_schema_hash").unwrap_or_default().to_owned();
1518    let actual_hash = container.schema.hash_hex();
1519    if declared_hash != actual_hash {
1520        problem(&mut report, format!("_schema_hash is {declared_hash}, the fields hash to {actual_hash}"));
1521    }
1522
1523    let mut frame_records_total: u64 = 0;
1524    // Positionally, never by id: ids repeat legitimately when a caller
1525    // partitions by hour and a spill closes under hour 0 after hour 23, and an
1526    // id-keyed sweep would read the first twin twice and never see the second.
1527    let frame_count = container.frames.len();
1528    for index in 0..frame_count {
1529        let frame = container
1530            .frames
1531            .get(index)
1532            .copied()
1533            .unwrap_or(Frame { id: 0, offset: 0, len: 0, hash: 0 });
1534        let frame_id = frame.id;
1535        report.frames = report.frames.saturating_add(1);
1536        if frame.hash != 0 {
1537            match read_at(
1538                &mut container.file,
1539                container.data_start.saturating_add(frame.offset),
1540                to_usize(frame.len)?,
1541            ) {
1542                Ok(bytes) => {
1543                    let actual = xxh64(&bytes);
1544                    if actual != frame.hash {
1545                        problem(
1546                            &mut report,
1547                            format!(
1548                                "frame {frame_id}: checksum mismatch, _frames says {:016x}, the bytes hash to {actual:016x}",
1549                                frame.hash
1550                            ),
1551                        );
1552                    }
1553                }
1554                Err(error) => problem(&mut report, format!("frame {frame_id}: {error}")),
1555            }
1556        }
1557        match container.read_frame_at(index) {
1558            Ok(raw) => {
1559                let (records, leftover) = split_records(&raw, container.schema.layout);
1560                if leftover > 0 {
1561                    problem(&mut report, format!("frame {frame_id}: {leftover} torn bytes"));
1562                }
1563                frame_records_total = frame_records_total.saturating_add(records.len() as u64);
1564                report.records = report.records.saturating_add(records.len() as u64);
1565                if let Some(callback) = check.as_mut() {
1566                    for record in records {
1567                        if let Some(text) = callback(record) {
1568                            problem(&mut report, format!("frame {frame_id}: {text}"));
1569                        }
1570                    }
1571                }
1572            }
1573            Err(error) => problem(&mut report, format!("frame {frame_id}: {error}")),
1574        }
1575    }
1576
1577    match container.read_tail() {
1578        Ok(raw) => {
1579            let (records, _leftover) = split_records(&raw, container.schema.layout);
1580            report.records = report.records.saturating_add(records.len() as u64);
1581            if let Some(callback) = check.as_mut() {
1582                for record in records {
1583                    if let Some(text) = callback(record) {
1584                        problem(&mut report, format!("tail: {text}"));
1585                    }
1586                }
1587            }
1588        }
1589        Err(error) => problem(&mut report, format!("tail: {error}")),
1590    }
1591
1592    // `_records` covers closed frames only; tail records are the not yet
1593    // closed remainder on top and never a mismatch.
1594    let declared_records: u64 = container.headers.get("_records").unwrap_or("0").parse().unwrap_or(0);
1595    if declared_records != frame_records_total {
1596        problem(
1597            &mut report,
1598            format!("_records says {declared_records}, the frames hold {frame_records_total}"),
1599        );
1600    }
1601    Ok(report)
1602}

1604/// What [`rebuild_headers`] found and did.
1605#[derive(Debug, Default)]
1606pub struct RebuildReport {
1607    /// Header values that disagreed with the data, `key: stored -> actual`.
1608    pub differences: Vec<String>,
1609    /// Whether the zone was rewritten.
1610    pub fixed: bool,
1611}

1613/// Rescans the data section by walking zstd frames, recomputes `_frames`,
1614/// `_records` and `_bytes_raw`, and rewrites the zone when `fix` is set.
1615///
1616/// Frame ids cannot be recovered from the bytes alone, so ids of frames
1617/// already present keep their stored value and newly discovered frames are
1618/// numbered on from the largest known id.
1619///
1620/// # Errors
1621/// Returns a message when the file cannot be opened or the fix cannot
1622/// write.
1623pub fn rebuild_headers(path: &Path, fix: bool) -> Result<RebuildReport> {
1624    replay_journal(path)?;
1625    let mut container = Container::open_inner(path, fix)?;
1626    let data_len = container.file_len.saturating_sub(container.data_start);
1627
1628    let mut actual_frames: Vec<Frame> = Vec::new();
1629    let mut records: u64 = 0;
1630    let mut bytes_raw: u64 = 0;
1631    let mut position = 0usize;
1632    let mut next_id = container
1633        .frames
1634        .iter()
1635        .map(|frame| frame.id.saturating_add(1))
1636        .max()
1637        .unwrap_or(0);
1638    while (position as u64) < data_len {
1639        let absolute = container.data_start.saturating_add(position as u64);
1640        if read_at(&mut container.file, absolute, 4)? != ZSTD_MAGIC.as_slice() {
1641            break;
1642        }
1643        // One frame at a time, read in growing windows rather than by loading
1644        // the whole data section: the format is meant for files that outgrow
1645        // memory, so a maintenance pass that does not is a maintenance pass
1646        // nobody can run on the files that need it.
1647        let (consumed, raw) = decode_frame_at(&mut container.file, absolute, data_len.saturating_sub(position as u64))?;
1648        // Recomputing the hash here is what upgrades a file written before the
1649        // field existed: `rebuild_headers(path, true)` rewrites `_frames` with
1650        // the fourth field filled in.
1651        let hash = xxh64(&read_at(&mut container.file, absolute, consumed)?);
1652        let (frame_records, leftover) = split_records(&raw, container.schema.layout);
1653        if leftover > 0 {
1654            return Err(malformed(format!(
1655                "frame at offset {position}: {leftover} torn bytes inside a closed frame"
1656            )));
1657        }
1658        records = records.saturating_add(frame_records.len() as u64);
1659        bytes_raw = bytes_raw.saturating_add(raw.len() as u64);
1660        let id = container.frames.get(actual_frames.len()).map_or_else(
1661            || {
1662                let id = next_id;
1663                next_id = next_id.saturating_add(1);
1664                id
1665            },
1666            |frame| frame.id,
1667        );
1668        actual_frames.push(Frame { id, offset: position as u64, len: consumed as u64, hash });
1669        position = position.saturating_add(consumed);
1670    }
1671    // The raw tail is deliberately left out of the counters: `_records`
1672    // and `_bytes_raw` publish closed frames only.
1673    let mut report = RebuildReport::default();
1674    let stored_frames = render_frames(&container.frames);
1675    let found_frames = render_frames(&actual_frames);
1676    if stored_frames != found_frames {
1677        report
1678            .differences
1679            .push(format!("_frames: {stored_frames:?} -> {found_frames:?}"));
1680    }
1681    let stored_records = container.headers.get("_records").unwrap_or("0").to_owned();
1682    if stored_records != records.to_string() {
1683        report
1684            .differences
1685            .push(format!("_records: {stored_records} -> {records}"));
1686    }
1687    let stored_raw = container.headers.get("_bytes_raw").unwrap_or("0").to_owned();
1688    if stored_raw != bytes_raw.to_string() {
1689        report
1690            .differences
1691            .push(format!("_bytes_raw: {stored_raw} -> {bytes_raw}"));
1692    }
1693
1694    if fix && !report.differences.is_empty() {
1695        container.frames = actual_frames;
1696        container.records = records;
1697        container.bytes_raw = bytes_raw;
1698        container.records_closed = records;
1699        container.bytes_closed = bytes_raw;
1700        container.headers.set("_frames", &found_frames)?;
1701        container.flush_headers()?;
1702        report.fixed = true;
1703    }
1704    Ok(report)
1705}

1707/// Reads exactly `len` bytes at an absolute offset, short reads included.
1708fn read_at(file: &mut File, offset: u64, len: usize) -> Result<Vec<u8>> {
1709    let mut buffer = vec![0u8; len];
1710    file.seek(SeekFrom::Start(offset)).map_err(Error::from)?;
1711    let mut filled = 0usize;
1712    while filled < len {
1713        let read = file
1714            .read(buffer.get_mut(filled..).unwrap_or_default())
1715            .map_err(Error::from)?;
1716        if read == 0 {
1717            buffer.truncate(filled);
1718            return Ok(buffer);
1719        }
1720        filled = filled.saturating_add(read);
1721    }
1722    Ok(buffer)
1723}
1724
1725/// Decodes the frame starting at `offset`, reading it in growing windows.
1726///
1727/// A zstd frame does not announce its compressed length, so the window starts
1728/// small and doubles until the frame decodes. The cost is re-reading and
1729/// re-decoding a frame at most twice over; the gain is that memory is bounded
1730/// by one frame instead of by the file.
1731fn decode_frame_at(file: &mut File, offset: u64, available: u64) -> Result<(usize, Vec<u8>)> {
1732    let mut window = 256usize * 1024;
1733    loop {
1734        let want = to_usize((window as u64).min(available))?;
1735        let chunk = read_at(file, offset, want)?;
1736        match decode_one_frame(&chunk) {
1737            Ok(result) => return Ok(result),
1738            Err(error) => {
1739                if chunk.len() as u64 >= available {
1740                    return Err(error);
1741                }
1742                window = window
1743                    .checked_mul(2)
1744                    .ok_or_else(|| malformed("frame is impossibly large"))?;
1745            }
1746        }
1747    }
1748}
1749
1750/// Decompresses one frame, refusing to produce more than
1751/// [`MAX_FRAME_RAW_BYTES`].
1752///
1753/// A compressor is free to turn a few kilobytes into terabytes, and a reader
1754/// that trusts it hands the process to whoever wrote the file.
1755fn decode_bounded(compressed: &[u8]) -> Result<Vec<u8>> {
1756    let (_consumed, raw) = decode_one_frame(compressed)?;
1757    Ok(raw)
1758}
1759
1760/// Decodes exactly one zstd frame, returning (consumed bytes, raw bytes).
1761fn decode_one_frame(input: &[u8]) -> Result<(usize, Vec<u8>)> {
1762    let mut context = zstd::zstd_safe::DCtx::create();
1763    let mut in_buffer = zstd::zstd_safe::InBuffer::around(input);
1764    let mut raw = Vec::new();
1765    let mut scratch = vec![0u8; 128 * 1024];
1766    loop {
1767        let mut out_buffer = zstd::zstd_safe::OutBuffer::around(scratch.as_mut_slice());
1768        let hint = context
1769            .decompress_stream(&mut out_buffer, &mut in_buffer)
1770            .map_err(|code| Error::Compression(zstd::zstd_safe::get_error_name(code).to_owned()))?;
1771        raw.extend_from_slice(out_buffer.as_slice());
1772        if raw.len() as u64 > MAX_FRAME_RAW_BYTES {
1773            return Err(malformed(format!("frame decompresses to more than {MAX_FRAME_RAW_BYTES} bytes")));
1774        }
1775        if hint == 0 {
1776            return Ok((in_buffer.pos(), raw));
1777        }
1778        if out_buffer.pos() == 0 && in_buffer.pos() == input.len() {
1779            return Err(malformed("truncated zstd frame"));
1780        }
1781    }
1782}

1784/// What [`repack`] measured.
1785#[derive(Debug, Default)]
1786pub struct RepackReport {
1787    /// The file size before, bytes.
1788    pub bytes_before: u64,
1789    /// The file size after, bytes.
1790    pub bytes_after: u64,
1791    /// Frames re-encoded.
1792    pub frames: usize,
1793}

1795/// Re-encodes every closed frame at `level`, transactionally per file.
1796///
1797/// The new version is assembled next to the original (`<file>.repack`) and
1798/// atomically renamed over it; a failure at any point leaves the original
1799/// untouched. Frame boundaries and ids survive; the raw tail, when present,
1800/// is carried over as is.
1801///
1802/// # Errors
1803/// Returns a message on any read, encode or write failure.
1804pub fn repack(path: &Path, level: i32) -> Result<RepackReport> {
1805    let area = peek_headers(path)?.0.header_area;
1806    repack_with_header_area(path, level, area)
1807}
1808
1809/// Re-encodes every closed frame at `level` and gives the rebuilt file a
1810/// header zone of `header_area` bytes.
1811///
1812/// This is the way out of [`Error::ZoneFull`]. The frame list grows with every
1813/// frame closed and the zone it lives in cannot move, so a file written to for
1814/// long enough eventually cannot record another frame. Repacking rebuilds the
1815/// file from scratch, which is the one moment the zone can be a different size.
1816/// [`max_frames_for`] says how far a given size goes.
1817///
1818/// The new version is assembled next to the original (`<file>.repack`) and
1819/// atomically renamed over it; a failure at any point leaves the original
1820/// untouched. Frames are re-encoded and written one at a time, so the memory
1821/// this needs is the size of the largest frame rather than the size of the
1822/// file. Frame boundaries and ids survive; the raw tail, when present, is
1823/// carried over as is.
1824///
1825/// # Errors
1826/// Returns a message on any read, encode or write failure, and
1827/// [`Error::ZoneFull`] when the headers do not fit the requested zone.
1828pub fn repack_with_header_area(path: &Path, level: i32, header_area: u32) -> Result<RepackReport> {
1829    replay_journal(path)?;
1830    if header_area == 0 || header_area > MAX_HEADER_AREA {
1831        return Err(usage(format!("header zone of {header_area} bytes is outside 1..={MAX_HEADER_AREA}")));
1832    }
1833    let mut container = Container::open_read(path)?;
1834    let mut report = RepackReport { bytes_before: container.file_len, ..RepackReport::default() };
1835
1836    let staging = {
1837        let mut name = path.file_name().map(ToOwned::to_owned).unwrap_or_default();
1838        name.push(".repack");
1839        path.with_file_name(name)
1840    };
1841    let preamble = Preamble { version: VERSION, flags: FLAG_COMPRESSED, header_area };
1842
1843    let result = (|| -> Result<(u64, usize)> {
1844        let mut out = File::create(&staging).map_err(io_at(&staging))?;
1845        out.write_all(&preamble.encode()).map_err(Error::from)?;
1846        // A placeholder of the right size: the real zone needs the frame list,
1847        // and the frame list needs the frames written. The zone has a fixed
1848        // size and a fixed position, so it can be filled in at the end.
1849        out.write_all(&vec![0u8; header_area as usize]).map_err(Error::from)?;
1850
1851        let mut new_frames: Vec<Frame> = Vec::new();
1852        let mut offset: u64 = 0;
1853        let frame_list = container.frames.clone();
1854        for frame in frame_list {
1855            let raw = container.read_frame(frame.id)?;
1856            let compressed = compress_frame(&raw, level)?;
1857            out.write_all(&compressed).map_err(Error::from)?;
1858            new_frames.push(Frame {
1859                id: frame.id,
1860                offset,
1861                len: compressed.len() as u64,
1862                hash: xxh64(&compressed),
1863            });
1864            offset = offset.saturating_add(compressed.len() as u64);
1865        }
1866        let frames_written = new_frames.len();
1867
1868        let tail = container.read_tail()?;
1869        out.write_all(&tail).map_err(Error::from)?;
1870
1871        let mut headers = container.headers.clone();
1872        headers.set("_frames", &render_frames(&new_frames))?;
1873        headers.set("_compression_level", &level.to_string())?;
1874        let zone = headers.render(header_area as usize)?;
1875        out.seek(SeekFrom::Start(PREAMBLE_BYTES as u64))
1876            .map_err(Error::from)?;
1877        out.write_all(&zone).map_err(Error::from)?;
1878        out.sync_all().map_err(Error::from)?;
1879
1880        let len = out.metadata().map_err(Error::from)?.len();
1881        Ok((len, frames_written))
1882    })();
1883
1884    match result {
1885        Ok((len, frames)) => {
1886            std::fs::rename(&staging, path).map_err(Error::from)?;
1887            sync_parent_dir(path);
1888            report.bytes_after = len;
1889            report.frames = frames;
1890            Ok(report)
1891        }
1892        Err(error) => {
1893            let _ignored = std::fs::remove_file(&staging);
1894            Err(error)
1895        }
1896    }
1897}

1899/// Roughly how many frames a header zone of `header_area` bytes can list.
1900///
1901/// The frame list is text, so the answer depends on how large the offsets
1902/// get: a file of a few megabytes spends fewer digits per frame than one of a
1903/// few terabytes. This assumes the largest offsets the given zone can lead to
1904/// and is therefore a floor, not an estimate — a real file fits at least this
1905/// many.
1906///
1907/// It exists because the alternative is finding the limit in production, on
1908/// the day a long-lived writer stops being able to close a frame.
1909#[must_use]
1910pub fn max_frames_for(header_area: u32) -> u64 {
1911    // The system headers other than `_frames`, generously: schema, source,
1912    // writer, timestamps, counters, and the preview at its full budget.
1913    const FIXED_OVERHEAD: u64 = 1600 + PREVIEW_BUDGET as u64;
1914    // `id,offset,len;` with room for 20-digit ids, offsets and lengths.
1915    const PER_FRAME: u64 = 64;
1916    u64::from(header_area)
1917        .saturating_sub(FIXED_OVERHEAD)
1918        .checked_div(PER_FRAME)
1919        .unwrap_or(0)
1920}
